cmake_minimum_required(VERSION 3.24)
project(OutbreakCityZero VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(SDL_SHARED OFF CACHE BOOL "" FORCE)
set(SDL_STATIC ON CACHE BOOL "" FORCE)
set(SDL_TESTS OFF CACHE BOOL "" FORCE)
set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(SDL3 GIT_REPOSITORY https://github.com/libsdl-org/SDL.git GIT_TAG release-3.2.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(SDL3)
add_executable(OutbreakCityZero src/main.cpp src/game.cpp)
target_include_directories(OutbreakCityZero PRIVATE include)
target_link_libraries(OutbreakCityZero PRIVATE SDL3::SDL3-static)
if(APPLE)
 set_target_properties(OutbreakCityZero PROPERTIES MACOSX_BUNDLE TRUE MACOSX_BUNDLE_BUNDLE_NAME "OUTBREAK: CITY ZERO" MACOSX_BUNDLE_GUI_IDENTIFIER "com.red5225.outbreakcityzero" MACOSX_BUNDLE_BUNDLE_VERSION "${PROJECT_VERSION}" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
 if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
  set_target_properties(OutbreakCityZero PROPERTIES XCODE_ATTRIBUTE_PRODUCT_BUNDLE_IDENTIFIER "com.red5225.outbreakcityzero" XCODE_ATTRIBUTE_IPHONEOS_DEPLOYMENT_TARGET "15.0" XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2" XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED "NO" XCODE_ATTRIBUTE_CODE_SIGNING_REQUIRED "NO")
 endif()
endif()

file(GLOB OUTBREAK_ASSETS CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/*.bmp")
target_sources(OutbreakCityZero PRIVATE ${OUTBREAK_ASSETS})
set_source_files_properties(${OUTBREAK_ASSETS} PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/assets")
add_custom_command(TARGET OutbreakCityZero POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "$<TARGET_FILE_DIR:OutbreakCityZero>/assets" COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets" "$<TARGET_FILE_DIR:OutbreakCityZero>/assets")
